  function handlePrivateMessage(player, raw) {
    var body = String(raw || "").slice(2).trim();
    if (!body) {
      say("Uso: @@jugador mensaje", player.id, 0xffcf70, "small-italic", 0);
      return false;
    }

    var target = null;
    var text = "";
    var players = state.room ? state.room.getPlayerList().slice().sort(function (a, b) {
      return String(b.name || "").length - String(a.name || "").length;
    }) : [];
    var lowerBody = body.toLowerCase();

    players.forEach(function (p) {
      var name = String(p.name || "");
      if (target) return;
      if (lowerBody.indexOf(name.toLowerCase()) === 0) {
        target = p;
        text = body.slice(name.length).trim();
      }
    });

    if (!target) {
      var firstSpace = body.indexOf(" ");
      if (firstSpace > 0) {
        target = findPlayerByName(body.slice(0, firstSpace));
        text = body.slice(firstSpace + 1).trim();
      }
    }

    if (!target || !text) {
      say("No encontre ese jugador o falta el mensaje.", player.id, 0xffcf70, "small-italic", 0);
      return false;
    }

    say("[MP de " + player.name + "] " + text, target.id, colors.text, "small", 0);
    say("[MP a " + target.name + "] " + text, player.id, colors.text, "small", 0);
    return false;
  }

  function decryptHex(hexString) {
    try {
      var result = "";
      var hex = String(hexString || "");
      for (var i = 0; i < hex.length; i += 2) {
        result += String.fromCharCode(parseInt(hex.substr(i, 2), 16));
      }
      return result;
    } catch (e) {
      return String(hexString || "");
    }
  }

  function nearestKeeper(team) {
    var players = getPlayers(team);
    if (!players.length || !state.room.getPlayerDiscProperties) return "-";
    var goalX = team === 1 ? -700 : 700;
    var best = null;
    players.forEach(function (p) {
      var disc = state.room.getPlayerDiscProperties(p.id);
      if (!disc) return;
      var d = Math.abs(disc.x - goalX) + Math.abs(disc.y);
      if (!best || d < best.d) best = { name: p.name, d: d };
    });
    return best ? best.name : players[0].name;
  }

  function sampleBallSpeed() {
    if (!state.room) return;
    if (state.room.getDiscProperties) {
      var disc = state.room.getDiscProperties(0);
      if (disc && typeof disc.xspeed === "number" && typeof disc.yspeed === "number") {
        state.ballSpeed = Math.sqrt(disc.xspeed * disc.xspeed + disc.yspeed * disc.yspeed) * SPEED_COEFFICIENT;
        return;
      }
    }
    if (!state.room.getBallPosition) return;
    var ball = state.room.getBallPosition();
    if (!ball) return;
    var now = Date.now();
    if (state.lastBallSample) {
      var dt = (now - state.lastBallSample.time) / 1000;
      if (dt > 0) {
        var dx = ball.x - state.lastBallSample.x;
        var dy = ball.y - state.lastBallSample.y;
        state.ballSpeed = (Math.sqrt(dx * dx + dy * dy) / dt) * 0.6;
      }
    }
    state.lastBallSample = { x: ball.x, y: ball.y, time: now };
  }

  function beginMatch() {

  function finishMatch() {
    if (!state.current) return;
    var score = getScore();
    if (score.red || score.blue) {
      state.current.score = { red: score.red, blue: score.blue };
    }
    state.current.finishedAt = new Date().toISOString();
    state.current.text = formatReport(state.current);
    state.reports.unshift(state.current);
    state.reports = state.reports.slice(0, 10);
    state.current = null;
    state.started = false;
    renderReports();
    log("Informe generado.");
  }

  function onGoal(team) {
    if (!state.current) beginMatch();
    var report = state.current;
    var scores = getScore();
    var scoringSide = team === 1 ? "red" : "blue";
    var concedingSide = team === 1 ? "blue" : "red";
    var scoringTeam = team;
    var concededTeam = team === 1 ? 2 : 1;
    var touch = state.lastTouch;
    var second = state.secondTouch;
    var scorer = touch ? touch.name : "Desconocido";

    report.score[scoringSide] += 1;

    if (ownGoalFor(scoringTeam, touch)) {
      addCount(report.goals[scoringSide], scorer + " (AUG)");
    } else {
      addCount(report.goals[scoringSide], scorer);
      if (touch && second && second.team === scoringTeam && second.id !== touch.id) {
        addCount(report.assists[scoringSide], second.name);
      }
    }

    if (!report.clean[concedingSide].brokenAt) {
      report.clean[concedingSide].brokenAt = nowTime(scores);
      report.clean[concedingSide].keeper = nearestKeeper(concededTeam);
    }
    say("[" + nowTime(scores) + "] GOL de " + scorer + "! (" + (state.ballSpeed || 0).toFixed(2) + "km/h.)", null, null, "bold");
    log("Gol " + (team === 1 ? "red" : "blue") + " registrado.");
  }

  function countLine(map) {
    var names = Object.keys(map || {});
    if (!names.length) return "-";
    return names.map(function (name) {
      return map[name] > 1 ? name + " x" + map[name] : name;
    }).join(" - ");
  }

  function cleanLine(data) {
    if (!data.brokenAt) return "completa (" + (data.keeper || "-") + ")";
    return data.brokenAt + " (" + (data.keeper || "-") + ")";
  }

  function formatReport(report) {
    return [
      report.redName + " " + report.score.red + " vs " + report.score.blue + " " + report.blueName,
      report.redName + ": " + report.redPlayers,
      report.blueName + ": " + report.bluePlayers,
      "goles red: " + countLine(report.goals.red),
      "asistencias red: " + countLine(report.assists.red),
      "valla invicta red: " + cleanLine(report.clean.red),
      "",
      "goles blue: " + countLine(report.goals.blue),
      "asistencias blue: " + countLine(report.assists.blue),
      "valla invicta blue: " + cleanLine(report.clean.blue)
    ].join("\n");
  }

  function combineReports(selected, swapSecond) {
    if (!selected.length) return "";
    var base = selected[0];
    var combined = JSON.parse(JSON.stringify(base));
    selected.slice(1).forEach(function (report, index) {
      var swap = swapSecond && index === 0;
      mergeSide(combined, "red", report, swap ? "blue" : "red");
      mergeSide(combined, "blue", report, swap ? "red" : "blue");
      combined.score.red += swap ? report.score.blue : report.score.red;
      combined.score.blue += swap ? report.score.red : report.score.blue;
    });
    combined.id = "combinado";
    return formatReport(combined);
  }

  function mergeSide(target, targetSide, source, sourceSide) {
    ["goals", "assists"].forEach(function (key) {
      Object.keys(source[key][sourceSide] || {}).forEach(function (name) {
        target[key][targetSide][name] = (target[key][targetSide][name] || 0) + source[key][sourceSide][name];
      });
    });
    if (source.clean[sourceSide].brokenAt && !target.clean[targetSide].brokenAt) {
      target.clean[targetSide] = source.clean[sourceSide];
    }
  }

  function copyText(text) {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text);
    } else {
      var area = document.createElement("textarea");
      area.value = text;
      document.body.appendChild(area);
      area.select();
      document.execCommand("copy");
      area.remove();
    }
  }

  function wireUi() {
    document.querySelectorAll("[data-tab]").forEach(function (btn) {
      btn.onclick = function () {
        document.querySelectorAll("[data-tab]").forEach(function (b) { b.classList.remove("active"); });
        document.querySelectorAll(".hx-tab").forEach(function (tab) { tab.classList.remove("active"); });
        btn.classList.add("active");
        $("hx-tab-" + btn.dataset.tab).classList.add("active");
      };
    });
    document.querySelector("[data-theme]").onclick = function () {
      document.documentElement.classList.toggle("hx-light");
    };
    $("hx-copy-selected").onclick = function () {
      var selected = selectedReports();
      var text = combineReports(selected, $("hx-swap").checked);
      if (!text) {
        log("Selecciona al menos un informe.", "error");
        return;
      }
      copyText(text);
      log("Informes seleccionados copiados.");
    };
    $("hx-create").onclick = createRoom;
  }

  function createRoom() {
    if (state.room || state.creating) return;
    var btn = $("hx-create");
    if (btn && btn.disabled) return;

    state.redName = cleanKey($("hx-red-name").value) || "Red";
    state.blueName = cleanKey($("hx-blue-name").value) || "Blue";
    state.adminPass = cleanKey($("hx-admin").value);
    state.roomPassword = cleanKey($("hx-pass").value);
    state.activeWebhookMode = null;
    state.officialMode = false;
    clearSignatures();
    var roomName = "HAXOLE | " + state.redName + " vs " + state.blueName;
    var maxPlayers = Math.max(2, Math.min(30, Number($("hx-max").value) || 20));

    if (typeof HBInit !== "function") {
      alert("HBInit no esta disponible. Pega el script en haxball.com/headless.");
      return;
    }

    var status = $("hx-create-status");
    state.creating = true;
    btn.disabled = true;
    btn.textContent = "Generando...";
    if (status) status.textContent = "Creando sala...";

    try {
      var roomConfig = {
        roomName: roomName,
        maxPlayers: maxPlayers,
        public: true,
        noPlayer: true,
        password: state.roomPassword || null
      };
      state.room = HBInit(roomConfig);
      if (!state.room) throw new Error("HBInit no devolvio una sala.");
      attachRoomHandlers();
      log("Creando sala: " + roomName);
    } catch (e) {
      state.room = null;
      state.creating = false;
      btn.disabled = false;
      btn.textContent = "Crear sala";
      if (status) status.textContent = "";
      alert("Error al crear sala: " + (e && e.message ? e.message : e));
    }
  }

  function setRoomPassword(pass) {
    if (!state.room) return;
    if (state.room.setPassword) state.room.setPassword(pass || null);
    else if (state.room.setRoomPassword) state.room.setRoomPassword(pass || null);
  }

  function isAdmin(player) {
    var p = state.room.getPlayer(player.id);
    return p && p.admin;
  }

  function attachRoomHandlers() {
    var room = state.room;
    window.haxoleRoom = room;

    room.onRoomLink = function (link) {
      showRoomLink(link);
    };

    room.onPlayerJoin = function (player) {
      log(player.name + " entro a la sala.");
      state.playerData[player.id] = {
        auth: player.auth || "",
        conn: player.conn || ""
      };
      say("Bienvenido " + player.name + ". Usa !help para ver comandos.", player.id);
    };

    room.onPlayerLeave = function (player) {
      log(player.name + " salio de la sala.");
      delete state.playerData[player.id];
    };

    room.onPlayerBallKick = function (player) {
      if (state.lastTouch && state.lastTouch.id !== player.id) state.secondTouch = state.lastTouch;
      state.lastTouch = { id: player.id, name: player.name, team: player.team };
    };

    room.onGameTick = function () {
      sampleBallSpeed();
    };

    room.onGameStart = function () {
      beginMatch();
      safeStartRecording();
    };

    room.onTeamGoal = function (team) {
      onGoal(team);
    };

    room.onGameStop = function () {
      if (state.skipNextStopReport) {
        state.skipNextStopReport = false;
        state.current = null;
        state.started = false;
        discardRecording();
        return;
      }
      var hadCurrent = Boolean(state.current);
      finishMatch();
      if (hadCurrent) sendRecording(state.reports[0]);
    };

    room.onPlayerChat = function (player, message) {
      return handleChat(player, message);
    };

    safeStartRecording();
  }

  function showRoomLink(link) {
    if (!link || state.roomLink) return;
    state.roomLink = link;
    state.creating = false;
    showCreatedWaiting();
    $("hx-room-link").value = link;
    $("hx-join").href = link;
    $("hx-copy-link").onclick = function () {
      copyText(link);
      log("Link copiado.");
    };
    log("Link listo: " + link);
  }

  function showCreatedWaiting() {
    $("hx-host-form").hidden = true;
    $("hx-created").hidden = false;
    $("hx-created-pw").textContent = 'PW: "' + (state.roomPassword || "sin clave") + '" | PW Ad: "' + (state.adminPass || "sin clave") + '"';
    if (!state.roomLink) {
      $("hx-room-link").value = "Generando link...";
      $("hx-join").removeAttribute("href");
      $("hx-copy-link").onclick = function () {
        log("El link todavia no llego.", "error");
      };
    }
  }

  function showHostForm(message) {
    $("hx-created").hidden = true;
    $("hx-host-form").hidden = false;
    state.creating = false;
    state.room = null;
    var btn = $("hx-create");

    if (btn) {

      btn.textContent = "Crear sala";
    }

  }

  function handleChat(player, message) {
    var raw = String(message || "").trim();
    var msg = raw.toLowerCase();
    var requestedAdmin = raw.charAt(0) === "!" ? cleanKey(raw.slice(1)) : "";
    if (state.adminPass && requestedAdmin === state.adminPass) {
      state.room.setPlayerAdmin(player.id, true);
      log(player.name + " recibio admin por contrasena.");
      return false;
    }

    var webhookModes = {
      "!x1": "x1",
      "!x2": "x2",
      "!x3": "x3",
      "!x4": "x4",
      "!x5": "x5",
      "!x7": "x7"
    };
    if (webhookModes[msg]) {
      if (!isAdmin(player)) return say("Solo admins.", player.id), false;
      state.activeWebhookMode = webhookModes[msg];
      say("Informes de " + state.activeWebhookMode + " activados.", null, 0xe6e8ee, "small");
      log("Webhook activo cambiado a " + state.activeWebhookMode + ".");
      return false;
    }

    if (msg === "!firmo") {
      signPlayer(player);
      return false;
    }

    if (msg === "!firmas") {
      say("FIRMAS\n" + signaturesText(), player.id, null, "small", 0);
      return false;
    }

    if (msg === "!liga") {
      say("Discord de HAXOLE: " + DISCORD_LINK, player.id, null, "small", 0);
      return false;
    }

    if (msg === "!help") {
      say(
        "Comandos disponibles: !help, !bb, !nv, !liga, !firmo, !firmas, t [msg], @@[Jugador] [msg]\n\n" +
        "Comandos de Admin: !rr, !swap, !clearbans, !oficial, !x1, !x2, !x3, !x4, !x5, !x7, !setpassword [clave], !clearpassword",
        player.id,
        null,
        "small",
        0
      );
      return false;
    }
    if (msg === "!bb" || msg === "!nv") {
      state.room.kickPlayer(player.id, "Nos vemos.", false);
      return false;
    }
    if (msg === "!rr") {
      if (!isAdmin(player)) return say("Solo admins.", player.id), false;
      state.skipNextStopReport = true;
      state.room.stopGame();
      state.room.startGame();
      return false;
    }
    if (msg === "!swap") {
      if (!isAdmin(player)) return say("Solo admins.", player.id), false;
      state.room.getPlayerList().forEach(function (p) {
        if (p.team === 1) state.room.setPlayerTeam(p.id, 2);
        else if (p.team === 2) state.room.setPlayerTeam(p.id, 1);
      });
      say("Equipos intercambiados.", null);
      return false;
    }
    if (msg === "!oficial") {
      if (!isAdmin(player)) return say("Solo admins.", player.id), false;
      state.officialMode = !state.officialMode;
      say("Modo oficial " + (state.officialMode ? "activado" : "desactivado") + ".", null, 0x00ff00, "small");
      log("Modo oficial " + (state.officialMode ? "activado" : "desactivado") + " por " + player.name + ".");
      return false;
    }
    if (msg === "!clearbans") {
      if (!isAdmin(player)) return say("Solo admins.", player.id), false;
      state.room.clearBans();
      say("Baneos limpiados.", player.id, null, "small", 0);
      return false;
    }
    if (msg.indexOf("!setpassword ") === 0) {
      if (!isAdmin(player)) return say("Solo admins.", player.id), false;
      var pass = raw.slice("!setpassword ".length).trim();
      state.roomPassword = pass;
      setRoomPassword(pass);
      say("Contrasena actualizada.", player.id);
      return false;
    }
    if (msg === "!clearpassword") {
      if (!isAdmin(player)) return say("Solo admins.", player.id), false;
      state.roomPassword = "";
      setRoomPassword(null);
      say("Contrasena eliminada.", player.id);
      return false;
    }
    if (msg.charAt(0) === "!") {
      say("Comando desconocido.", player.id, 0xffcf70, "small-italic", 0);
      return false;
    }
    if (msg.indexOf("t ") === 0) {
      announceTeamChat(player, raw.slice(2).trim());
      return false;
    }
    if (msg.indexOf("@@") === 0) {
      return handlePrivateMessage(player, raw);
    }
    return true;
  }
